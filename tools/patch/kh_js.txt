/* ===================== KAHOOT-STYLE REVIEW (self-contained mode) ===================== */
var khSession=null;
var KH_SHAPES=['\u25B2','\u25C6','\u25CF','\u25A0'];
function khHideAll(){['randomSetup','topicList','moduleList','moduleSetup','labelingList','ecgHome','ecgPick','ecgQuiz','ecgResults','kahootHome','kahootPlay','kahootResults'].forEach(hide);}
function khHome(){khHideAll();show('homeMenu');window.scrollTo(0,0);}
function showKahoot(){
  khHideAll();show('homeMenu');show('kahootHome');
  var box=el('kahootUnits');box.innerHTML='';
  allModules().forEach(function(m){
    unitsOfModule(m).forEach(function(u){
      var pool=BANK.filter(function(q){return q.module===m&&q.unit===u;});
      if(pool.length<4)return;
      var b=document.createElement('button');b.className='topic-btn';
      b.innerHTML='<span>'+unitLabel(u,m)+'</span><span class="cnt">'+pool.length+'</span>';
      b.onclick=function(){khBegin(m,u);};box.appendChild(b);
    });
  });
  window.scrollTo(0,0);
}
function khBegin(m,u){
  var pool=BANK.filter(function(q){return q.module===m&&q.unit===u;});
  var n=parseInt(el('khcount').value,10);if(isNaN(n)||n<1)n=10;if(n>pool.length)n=pool.length;
  khSession={m:m,u:u,queue:shuffle(pool.slice()).slice(0,n),idx:0,score:0,missed:[]};
  khHideAll();hide('homeMenu');show('kahootPlay');khRender();
}
/* Answer bank: correct answer + 3 distractors pulled from OTHER same-unit questions' answers */
function khOptions(qq){
  var correct=qq.o[qq.a];
  var bank=[];
  BANK.forEach(function(other){
    if(other===qq)return;
    if(other.module===qq.module&&other.unit===qq.unit){
      var c=other.o[other.a];
      if(c&&c!==correct&&bank.indexOf(c)===-1&&qq.o.indexOf(c)===-1)bank.push(c);
    }
  });
  var distract=shuffle(bank.slice()).slice(0,3);
  /* fallback: not enough same-unit bank answers -> use the question's own wrong options */
  if(distract.length<3){
    qq.o.forEach(function(o,i){if(i!==qq.a&&distract.length<3&&distract.indexOf(o)===-1&&o!==correct)distract.push(o);});
  }
  var opts=distract.concat([correct]);
  return shuffle(opts);
}
function khRender(){
  var S=khSession,qq=S.queue[S.idx];
  el('khProgress').textContent='Question '+(S.idx+1)+' of '+S.queue.length+'  \u2022  '+unitLabel(S.u,S.m);
  el('khScore').textContent='Score: '+S.score;
  var strip=el('khStrip');strip.innerHTML='';
  var qt=document.createElement('div');qt.textContent=qq.q;strip.appendChild(qt);
  var imgSrc=qq.img||qq.dimg;
  if(imgSrc){var im=document.createElement('img');im.src=imgSrc;im.alt='question image';strip.appendChild(im);}
  var grid=el('khGrid');grid.innerHTML='';
  S.currentOpts=khOptions(qq);
  S.currentOpts.forEach(function(opt,i){
    var b=document.createElement('button');b.className='kh-ans k'+i;b.disabled=false;
    var sh=document.createElement('span');sh.className='kh-shape';sh.textContent=KH_SHAPES[i];
    var tx=document.createElement('span');tx.textContent=opt;
    b.appendChild(sh);b.appendChild(tx);
    b.onclick=function(){khPick(i,b);};
    grid.appendChild(b);
  });
  var bn=el('khBanner');bn.className='kh-banner';bn.innerHTML='';
  el('khNextBtn').classList.add('hidden');
  window.scrollTo(0,0);
}
function khPick(i,btn){
  var S=khSession,qq=S.queue[S.idx],correct=qq.o[qq.a];
  var picked=S.currentOpts[i],right=(picked===correct);
  var buttons=el('khGrid').children;
  for(var k=0;k<buttons.length;k++){
    buttons[k].disabled=true;
    if(S.currentOpts[k]===correct)buttons[k].classList.add('kh-right');
  }
  if(!right)btn.classList.add('kh-wrongpick');
  if(right){S.score++;el('khScore').textContent='Score: '+S.score;}
  else S.missed.push(qq);
  var bn=el('khBanner');
  bn.className='kh-banner show '+(right?'ok':'no');
  bn.innerHTML=(right?'\u2713 Correct!':'\u2717 Not quite \u2014 correct answer: <b>'+correct+'</b>')+'<div class="kh-exp">'+qq.e+'</div>';
  el('khNextBtn').classList.remove('hidden');
  el('khNextBtn').textContent=(S.idx<S.queue.length-1)?'Next \u25B6':'See results \u25B6';
}
function khNext(){
  var S=khSession;
  if(S.idx<S.queue.length-1){S.idx++;khRender();}else khResults();
}
function khResults(){
  var S=khSession;khHideAll();hide('homeMenu');show('kahootResults');
  var pct=Math.round(S.score/S.queue.length*100);
  el('khPct').textContent=pct+'%';el('khRaw').textContent=S.score+' / '+S.queue.length;
  el('khCircle').className='score-circle '+(pct>=80?'pass':'fail');
  var v=el('khVerdict');v.className='verdict-big '+(pct>=80?'pass':'fail');
  v.textContent=pct>=80?'Great round!':'Keep drilling this unit';
  var box=el('khMissed');box.innerHTML='';
  if(S.missed.length){
    box.innerHTML='<h3>Review these</h3>';
    S.missed.forEach(function(qq){
      var d=document.createElement('div');d.className='missed';
      d.innerHTML='<div class="mq">'+qq.q+'</div><div class="ans good">\u2713 '+qq.o[qq.a]+'</div><div class="exp">'+qq.e+'</div>';
      box.appendChild(d);
    });
  }
  window.scrollTo(0,0);
}
function khQuit(){if(khSession&&khSession.idx>0){khResults();}else khHome();}
